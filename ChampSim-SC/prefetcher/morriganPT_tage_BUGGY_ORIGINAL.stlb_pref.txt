// ─────────────────────────────────────────────────────────────────────────────
// T-IRIP: TAGE-inspired history tables (OUR NOVEL CONTRIBUTION)
//
// Design:
//   Three tables indexed by different history lengths (H2, H4, H8).
//   Index = hash32(partial_vpn XOR folded_history(N))
//   Tag   = hash32(partial_vpn XOR (folded_history(N) >> 3)) & TAG_MASK
//   Selection: longest matching history wins (H8 > H4 > H2)
//   Confidence >= 1 required to issue prefetch
//   Falls back to original IRIP if no TAGE table matches
// ─────────────────────────────────────────────────────────────────────────────

// T-IRIP table sizes
#define THT_H2_SETS  128
#define THT_H2_ASSOC 4
#define THT_H4_SETS  128
#define THT_H4_ASSOC 4
#define THT_H8_SETS  64
#define THT_H8_ASSOC 4

// Maximum history length to track
#define TAGE_MAX_HIST 8

// T-IRIP table entry
class TAGE_ENTRY {
public:
    uint64_t tag;        // partial tag for collision detection
    int64_t  delta;      // predicted successor delta
    int      confidence; // saturating counter (0-3)
    uint64_t timestamp;  // for LRU replacement
    TAGE_ENTRY() : tag(0), delta(0), confidence(0), timestamp(0) {}
};

// history_reg[0] = most recent miss

int      history_len = 0;  // how many entries are valid

uint32_t hash32(uint32_t a){ 
    a = (a+0x479ab41d) + (a<<8);
    a = (a^0xe4aa10ce) ^ (a>>5);
    a = (a+0x9942f0a6) - (a<<14);
    a = (a^0x5aedd67d) ^ (a>>3);
    a = (a+0x17bea992) + (a<<7);
    return a;
}

int random_policy(int id){
    int range_from = 0;
    int range_to = -100;

    if(id == 1)
        range_to = PT_S1_ASSOC - 1;
    else if(id == 2)
        range_to = PT_S2_ASSOC - 1;
    else if(id == 4)
        range_to = PT_S4_ASSOC - 1;
    else if(id == 8)
        range_to = PT_S8_ASSOC - 1;
    else
        assert(0);

    random_device rand_dev;
    mt19937 generator(rand_dev());
    uniform_int_distribution<int> distr(range_from, range_to);
    return distr(generator);
}

// Compute folded XOR history fingerprint of length `len`
// Rotates each history entry by its position to avoid cancellation
static uint64_t fold_history(int len) {
    uint64_t h = 0;
    int n = (len < history_len) ? len : history_len;
    for (int i = 0; i < n; i++) {
        uint64_t v = history_reg[i] & TAG_MASK;
        // Rotate left by (i & 7) bits within 16-bit width
        int rot = i & 7;
        v = ((v << rot) | (v >> (16 - rot))) & TAG_MASK;
        h ^= v;
    }
    return h;
}

// Compute table index for a TAGE table of given size
static int tage_index(uint64_t partial_vpn, int hlen, int tsets, int tassoc) {
    uint64_t h   = fold_history(hlen);
    uint32_t raw = hash32((uint32_t)(partial_vpn ^ h));
    int num_sets = tsets / tassoc;
    int bits     = (int)ceil(log2(num_sets));
    return (int)(raw & ((1 << bits) - 1));
}

// Compute partial tag for a TAGE table entry
static uint64_t tage_tag(uint64_t partial_vpn, int hlen) {
    uint64_t h = fold_history(hlen);
    return hash32((uint32_t)(partial_vpn ^ (h >> 3))) & TAG_MASK;
}

// LRU victim selection for TAGE tables
static int tage_lru_victim(TAGE_ENTRY* row, int assoc) {
    int v = 0;
    for (int i = 1; i < assoc; i++)
        if (row[i].timestamp < row[v].timestamp) v = i;
    return v;
}

// Search a TAGE table row for a matching tag
static int tage_search(TAGE_ENTRY* row, int assoc, uint64_t tag) {
    for (int i = 0; i < assoc; i++)
        if (row[i].tag == tag && row[i].delta != 0) return i;
    return -1;
}

// ─────────────────────────────────────────────────────────────────────────────
// T-IRIP lookup: check H8 → H4 → H2 (longest history wins)
// Returns predicted successor VPN, or 0 if no match.
// ─────────────────────────────────────────────────────────────────────────────
static int64_t tage_lookup(uint64_t partial_vpn) {
    // H8 check (longest history = highest priority)
    if (history_len >= 2) {
        int idx8 = tage_index(partial_vpn, 8, THT_H8_SETS, THT_H8_ASSOC);
        uint64_t t8 = tage_tag(partial_vpn, 8);
        int w8 = tage_search(tht_h8[idx8], THT_H8_ASSOC, t8);
        if (w8 >= 0 && tht_h8[idx8][w8].confidence >= 1) {
            tht_h8[idx8][w8].timestamp = s_timer;
            return tht_h8[idx8][w8].delta;
        }

        // H4 check
        int idx4 = tage_index(partial_vpn, 4, THT_H4_SETS, THT_H4_ASSOC);
        uint64_t t4 = tage_tag(partial_vpn, 4);
        int w4 = tage_search(tht_h4[idx4], THT_H4_ASSOC, t4);
        if (w4 >= 0 && tht_h4[idx4][w4].confidence >= 1) {
            tht_h4[idx4][w4].timestamp = s_timer;
            return tht_h4[idx4][w4].delta;
        }

        // H2 check (shortest history = lowest priority)
        int idx2 = tage_index(partial_vpn, 2, THT_H2_SETS, THT_H2_ASSOC);
        uint64_t t2 = tage_tag(partial_vpn, 2);
        int w2 = tage_search(tht_h2[idx2], THT_H2_ASSOC, t2);
        if (w2 >= 0 && tht_h2[idx2][w2].confidence >= 1) {
            tht_h2[idx2][w2].timestamp = s_timer;
            return tht_h2[idx2][w2].delta;
        }
    }
    return 0; // no match
}

// ─────────────────────────────────────────────────────────────────────────────
// T-IRIP update: record prev_vpn → curr_vpn edge in all history tables
// Called after we have both previous and current VPN.
// ─────────────────────────────────────────────────────────────────────────────
static void tage_update(uint64_t prev_partial, int64_t delta) {
    // Update all three tables with the same edge
    auto update_table = [&](TAGE_ENTRY table[][4], int sets, int assoc, int hlen) {
        if (history_len < 1) return;
        int      idx  = tage_index(prev_partial, hlen, sets, assoc);
        uint64_t tag  = tage_tag(prev_partial, hlen);
        int      way  = tage_search(table[idx], assoc, tag);

        if (way >= 0) {
            // Entry exists
            if (table[idx][way].delta == delta) {
                // Correct prediction: increment confidence (saturate at 3)
                if (table[idx][way].confidence < 3)
                    table[idx][way].confidence++;
            } else {
                // Wrong prediction: decrement confidence
                if (table[idx][way].confidence > 0)
                    table[idx][way].confidence--;
                // If confidence drops to 0, replace with new delta
                if (table[idx][way].confidence == 0) {
                    table[idx][way].delta      = delta;
                    table[idx][way].confidence = 1;
                    table[idx][way].timestamp  = s_timer;
                }
            }
        } else {
            // Install new entry: find LRU victim
            int victim = tage_lru_victim(table[idx], assoc);
            table[idx][victim].tag        = tag;
            table[idx][victim].delta      = delta;
            table[idx][victim].confidence = 1;
            table[idx][victim].timestamp  = s_timer;
        }
    };

    update_table(tht_h2, THT_H2_SETS, THT_H2_ASSOC, 2);
    update_table(tht_h4, THT_H4_SETS, THT_H4_ASSOC, 4);
    update_table(tht_h8, THT_H8_SETS, THT_H8_ASSOC, 8);
}

// ─────────────────────────────────────────────────────────────────────────────
// Free prefetching (page table locality) — unchanged from paper
// ─────────────────────────────────────────────────────────────────────────────
void CACHE::free_prefetching(uint64_t ip, uint64_t addr, int cache_line_position_n,
                              uint64_t pf_addr, int* free_indexes, uint64_t instr_id,
                              int type, int iflag) {
    int fi_lim = -100;
    for (int fi = 13; fi >= 0; fi--) {
        fi_lim = free_indexes[fi] + cache_line_position_n;
        if ((fi_lim >= 0) && (fi_lim <= 7))
            prefetch_page(ip, addr, pf_addr + free_indexes[fi], FILL_L2, 0, 1, 1,
                          free_indexes[fi], instr_id, type, iflag, 1, 0, 1);
    }
}

// ─────────────────────────────────────────────────────────────────────────────
// Initialize
// ─────────────────────────────────────────────────────────────────────────────
void CACHE::stlb_prefetcher_initialize() {
    cout << "CPU " << cpu << " STLB -- Morrigan + T-IRIP (TAGE) I-TLB Prefetcher" << endl;
    s_timer      = 0;
    previous_vpn = 0;
    history_len  = 0;

    // Clear T-IRIP tables
    int s2 = THT_H2_SETS / THT_H2_ASSOC;
    for (int i = 0; i < s2; i++)
        for (int j = 0; j < THT_H2_ASSOC; j++)
            tht_h2[i][j] = TAGE_ENTRY();

    int s4 = THT_H4_SETS / THT_H4_ASSOC;
    for (int i = 0; i < s4; i++)
        for (int j = 0; j < THT_H4_ASSOC; j++)
            tht_h4[i][j] = TAGE_ENTRY();

    int s8 = THT_H8_SETS / THT_H8_ASSOC;
    for (int i = 0; i < s8; i++)
        for (int j = 0; j < THT_H8_ASSOC; j++)
            tht_h8[i][j] = TAGE_ENTRY();

    for (int i = 0; i < TAGE_MAX_HIST; i++)
        history_reg[i] = 0;

    tage_predictions   = 0;
    irip_predictions   = 0;
    total_istlb_misses = 0;
}

// ─────────────────────────────────────────────────────────────────────────────
// Main prefetcher operation — called on every STLB miss
// ─────────────────────────────────────────────────────────────────────────────
void CACHE::stlb_prefetcher_operate(uint64_t addr, uint64_t ip, uint8_t cache_hit,
                                     uint8_t type, int answer, int warmup,
                                     int* free_indexes, uint64_t instr_id, int iflag) {
    if (!iflag) assert(0);

    s_timer++;
    total_istlb_misses++;

    uint64_t current_vpn  = addr;
    uint64_t partial_vpn  = current_vpn & TAG_MASK;
    uint64_t clp          = 0;
    int cache_line_pos, bits, index, way = -1, victim, ignore, hit_level = -1;
    int free_bit = 0, free_distance = 0;

    // ── Confidence counter update on PB hit (answer != -1) ───────────────────
    if (answer != -1) {
        int ss = PT_S2_SETS / PT_S2_ASSOC;
        for (int i = 0; i < ss; ++i)
            for (int j = 0; j < PT_S2_ASSOC; ++j)
                for (int k = 0; k < 2; ++k)
                    if (irip_s2[i][j].successor_delta[k] != 0)
                        if (partial_vpn == ((irip_s2[i][j].successor_delta[k] + irip_s2[i][j].vpn) & TAG_MASK))
                            if (irip_s2[i][j].confidence[k] < pow(2, CNF_BITS)) {
                                irip_s2[i][j].confidence[k]++; break;
                            }

        ss = PT_S4_SETS / PT_S4_ASSOC;
        for (int i = 0; i < ss; ++i)
            for (int j = 0; j < PT_S4_ASSOC; ++j)
                for (int k = 0; k < 4; ++k)
                    if (irip_s4[i][j].successor_delta[k] != 0)
                        if (partial_vpn == ((irip_s4[i][j].successor_delta[k] + irip_s4[i][j].vpn) & TAG_MASK))
                            if (irip_s4[i][j].confidence[k] < pow(2, CNF_BITS)) {
                                irip_s4[i][j].confidence[k]++; break;
                            }

        ss = PT_S8_SETS / PT_S8_ASSOC;
        for (int i = 0; i < ss; ++i)
            for (int j = 0; j < PT_S8_ASSOC; ++j)
                for (int k = 0; k < 8; ++k)
                    if (irip_s8[i][j].successor_delta[k] != 0)
                        if (partial_vpn == ((irip_s8[i][j].successor_delta[k] + irip_s8[i][j].vpn) & TAG_MASK))
                            if (irip_s8[i][j].confidence[k] < pow(2, CNF_BITS)) {
                                irip_s8[i][j].confidence[k]++; break;
                            }
    }

    // ── STEP 1: T-IRIP lookup (our novel contribution) ────────────────────────
    // Check TAGE history tables BEFORE falling to IRIP.
    // Longer history = more specific context = higher priority.
    bool tage_hit = false;
    if (history_len >= 2) {
        int64_t tage_delta = tage_lookup(partial_vpn);
        if (tage_delta != 0) {
            uint64_t pf_page = current_vpn + tage_delta;

            free_bit      = 0;
            free_distance = 0;
            cache_line_pos = (int)(current_vpn & 0x07);

            if (tage_delta > 0) {
                if (tage_delta + cache_line_pos < 8) {
                    free_bit = 1;
                    free_distance = (int)(tage_delta + cache_line_pos);
                }
            } else {
                if (cache_line_pos - tage_delta < 8) {
                    free_bit = 1;
                    free_distance = (int)(cache_line_pos - tage_delta);
                }
            }

            // Spatial prefetching for T-IRIP prediction
            if (ENABLE_PREF_FP) {
                clp = (pf_page & 0x07);
                free_prefetching(ip, current_vpn, (int)clp, pf_page,
                                 free_indexes, instr_id, type, iflag);
            }

            prefetch_page(ip, current_vpn, pf_page, FILL_L2, 0,
                          free_bit, free_bit, free_distance,
                          instr_id, type, iflag, 0, 0, 1);

            tage_hit = true;
            tage_predictions++;
        }
    }

    // ── STEP 2: IRIP lookup (original Morrigan — unchanged) ──────────────────
    bits  = (int)ceil(log2(PT_S1_SETS / PT_S1_ASSOC));
    index = hash32(partial_vpn) & ((1 << bits) - 1);
    way   = search_markov_table(index, partial_vpn, 1);

    if (way == -1) {
        bits  = (int)ceil(log2(PT_S2_SETS / PT_S2_ASSOC));
        index = hash32(partial_vpn) & ((1 << bits) - 1);
        way   = search_markov_table(index, partial_vpn, 2);
        if (way != -1)
            hit_level = 2;
        else {
            bits  = (int)ceil(log2(PT_S4_SETS / PT_S4_ASSOC));
            index = hash32(partial_vpn) & ((1 << bits) - 1);
            way   = search_markov_table(index, partial_vpn, 4);
            if (way != -1)
                hit_level = 4;
            else {
                bits  = (int)ceil(log2(PT_S8_SETS / PT_S8_ASSOC));
                index = hash32(partial_vpn) & ((1 << bits) - 1);
                way   = search_markov_table(index, partial_vpn, 8);
                if (way != -1)
                    hit_level = 8;
            }
        }
    } else
        hit_level = 1;

    if (way == -1) {
        // IRIP miss: SDP fallback (spatial/sequential)
        int fi_lim = -100;
        cache_line_pos = (int)(current_vpn & 0x07);

        if (answer == -1 && ENABLE_FP == 1)
            ignore = 1;
        else {
            int free = 0;
            if (answer == -1) free = 1;
            for (int fi = 13; fi >= 0; fi--) {
                fi_lim = free_indexes[fi] + cache_line_pos;
                if ((fi_lim >= 0) && (fi_lim <= 7)) {
                    prefetch_page(ip, addr, current_vpn + free_indexes[fi], FILL_L2, 0,
                                  free, free, free_indexes[fi], instr_id, type, iflag, 0, 0, 0);
                    if (free == 0) free = 1;
                }
            }
        }

        int bits_1   = (int)ceil(log2(PT_S1_SETS / PT_S1_ASSOC));
        int index_1  = hash32(partial_vpn) & ((1 << bits_1) - 1);
        int victim_1 = search_markov_table(index_1, 0, 1);
        if (victim_1 == -1) {
            if      (RP_MP == 0) victim_1 = lru_policy(index_1, 1);
            else if (RP_MP == 1) victim_1 = lfu_policy(index_1, 1);
            else if (RP_MP == 2) victim_1 = random_policy(1);
            else { cout << "Unknown RP_MP" << endl; assert(0); }
        }
        irip_s1[index_1][victim_1].vpn             = partial_vpn;
        irip_s1[index_1][victim_1].timestamp        = s_timer;
        irip_s1[index_1][victim_1].freq             = 0;
        irip_s1[index_1][victim_1].successor_delta  = 0;

    } else {
        // IRIP hit: issue prediction(s)
        uint64_t pf_page = 0;
        free_bit = 0; free_distance = 0;
        if (!tage_hit) irip_predictions++;

        if (hit_level == 1) {
            irip_s1[index][way].timestamp = s_timer;
            irip_s1[index][way].freq++;

            if (irip_s1[index][way].successor_delta != 0) {
                if (answer == -1) {
                    cache_line_pos = (int)(current_vpn & 0x07);
                    if (irip_s1[index][way].successor_delta > 0) {
                        if (irip_s1[index][way].successor_delta + cache_line_pos < 8) {
                            free_bit = 1;
                            free_distance = (int)(irip_s1[index][way].successor_delta + cache_line_pos);
                        }
                    } else {
                        if (cache_line_pos - irip_s1[index][way].successor_delta < 8) {
                            free_bit = 1;
                            free_distance = (int)(cache_line_pos - irip_s1[index][way].successor_delta);
                        }
                    }
                }
                pf_page = current_vpn + irip_s1[index][way].successor_delta;
                if (ENABLE_PREF_FP) {
                    clp = (pf_page & 0x07);
                    free_prefetching(ip, current_vpn, (int)clp, pf_page, free_indexes, instr_id, type, iflag);
                }
                prefetch_page(ip, current_vpn, pf_page, FILL_L2, 0,
                              free_bit, free_bit, free_distance, instr_id, type, iflag, 0, 0, 1);
            }
        } else {
            int max_conf = 0, max_conf_pos = 0;

            if (hit_level == 2) {
                irip_s2[index][way].timestamp = s_timer;
                irip_s2[index][way].freq++;
                max_conf = irip_s2[index][way].confidence[0]; max_conf_pos = 0;
                for (int l = 1; l < 2; ++l)
                    if (irip_s2[index][way].confidence[l] > max_conf) { max_conf = irip_s2[index][way].confidence[l]; max_conf_pos = l; }
            } else if (hit_level == 4) {
                irip_s4[index][way].timestamp = s_timer;
                irip_s4[index][way].freq++;
                max_conf = irip_s4[index][way].confidence[0]; max_conf_pos = 0;
                for (int l = 1; l < 4; ++l)
                    if (irip_s4[index][way].confidence[l] > max_conf) { max_conf = irip_s4[index][way].confidence[l]; max_conf_pos = l; }
            } else if (hit_level == 8) {
                irip_s8[index][way].timestamp = s_timer;
                irip_s8[index][way].freq++;
                max_conf = irip_s8[index][way].confidence[0]; max_conf_pos = 0;
                for (int l = 1; l < 8; ++l)
                    if (irip_s8[index][way].confidence[l] > max_conf) { max_conf = irip_s8[index][way].confidence[l]; max_conf_pos = l; }
            } else assert(0);

            for (int k = 0; k < hit_level; ++k) {
                int64_t sd = 0;
                if      (hit_level == 2) sd = irip_s2[index][way].successor_delta[k];
                else if (hit_level == 4) sd = irip_s4[index][way].successor_delta[k];
                else if (hit_level == 8) sd = irip_s8[index][way].successor_delta[k];

                if (sd != 0) {
                    free_bit = 0; free_distance = 0;
                    cache_line_pos = (int)(current_vpn & 0x07);
                    if (sd > 0) {
                        if (sd + cache_line_pos < 8) { free_bit = 1; free_distance = (int)(sd + cache_line_pos); }
                    } else {
                        if (cache_line_pos - sd < 8) { free_bit = 1; free_distance = (int)(cache_line_pos - sd); }
                    }
                    pf_page = current_vpn + sd;
                    if (ENABLE_PREF_FP && (max_conf_pos == k)) {
                        clp = (pf_page & 0x07);
                        free_prefetching(ip, current_vpn, (int)clp, pf_page, free_indexes, instr_id, type, iflag);
                    }
                    prefetch_page(ip, current_vpn, pf_page, FILL_L2, 0,
                                  free_bit, free_bit, free_distance, instr_id, type, iflag, 0, 0, 1);
                }
            }
        }
    }

    // ── Update IRIP tables with prev → curr edge ──────────────────────────────
    if (previous_vpn != 0) {
        int bits_current, index_current, way_current;
        hit_level = -100;

        bits  = (int)ceil(log2(PT_S1_SETS / PT_S1_ASSOC));
        index = hash32((previous_vpn & TAG_MASK)) & ((1 << bits) - 1);
        way   = search_markov_table(index, (previous_vpn & TAG_MASK), 1);
        if (way == -1) {
            bits  = (int)ceil(log2(PT_S2_SETS / PT_S2_ASSOC));
            index = hash32((previous_vpn & TAG_MASK)) & ((1 << bits) - 1);
            way   = search_markov_table(index, (previous_vpn & TAG_MASK), 2);
            if (way != -1) hit_level = 2;
            else {
                bits  = (int)ceil(log2(PT_S4_SETS / PT_S4_ASSOC));
                index = hash32((previous_vpn & TAG_MASK)) & ((1 << bits) - 1);
                way   = search_markov_table(index, (previous_vpn & TAG_MASK), 4);
                if (way != -1) hit_level = 4;
                else {
                    bits  = (int)ceil(log2(PT_S8_SETS / PT_S8_ASSOC));
                    index = hash32((previous_vpn & TAG_MASK)) & ((1 << bits) - 1);
                    way   = search_markov_table(index, (previous_vpn & TAG_MASK), 8);
                    if (way != -1) hit_level = 8;
                }
            }
        } else hit_level = 1;

        int64_t delta  = (int64_t)current_vpn - (int64_t)previous_vpn;
        int bypass = 0, placed = 0;

        if (hit_level == 1) {
            if (irip_s1[index][way].successor_delta == 0)
                irip_s1[index][way].successor_delta = delta;
            else if (delta != irip_s1[index][way].successor_delta) {
                bits_current  = (int)ceil(log2(PT_S2_SETS / PT_S2_ASSOC));
                index_current = hash32((previous_vpn & TAG_MASK)) & ((1 << bits_current) - 1);
                way_current   = search_markov_table(index_current, 0, 2);
                if (way_current == -1) {
                    if      (RP_MP == 0) way_current = lru_policy(index_current, 2);
                    else if (RP_MP == 1) way_current = lfu_policy(index_current, 2);
                    else if (RP_MP == 2) way_current = random_policy(2);
                    else { cout << "Unknown RP" << endl; assert(0); }
                }
                irip_s2[index_current][way_current].vpn              = previous_vpn & TAG_MASK;
                irip_s2[index_current][way_current].successor_delta[0]= irip_s1[index][way].successor_delta;
                irip_s2[index_current][way_current].successor_delta[1]= delta;
                irip_s2[index_current][way_current].confidence[0]    = 0;
                irip_s2[index_current][way_current].confidence[1]    = 0;
                irip_s2[index_current][way_current].freq             = irip_s1[index][way].freq;
                irip_s2[index_current][way_current].timestamp        = irip_s1[index][way].timestamp;
                irip_s1[index][way].timestamp = 0; irip_s1[index][way].freq = 0;
                irip_s1[index][way].vpn = 0; irip_s1[index][way].successor_delta = 0;
            }
        } else if (hit_level == 2) {
            for (int j = 0; j < 2; ++j) if (irip_s2[index][way].successor_delta[j] == delta) { bypass = 1; break; }
            if (!bypass) {
                for (int j = 0; j < 2; ++j)
                    if (irip_s2[index][way].successor_delta[j] == 0) {
                        irip_s2[index][way].successor_delta[j] = delta;
                        irip_s2[index][way].confidence[j] = 0;
                        placed = 1; break;
                    }
                if (!placed) {
                    bits_current  = (int)ceil(log2(PT_S4_SETS / PT_S4_ASSOC));
                    index_current = hash32((previous_vpn & TAG_MASK)) & ((1 << bits_current) - 1);
                    way_current   = search_markov_table(index_current, (previous_vpn & TAG_MASK), 4);
                    if (way_current == -1) {
                        if      (RP_MP == 0) way_current = lru_policy(index_current, 4);
                        else if (RP_MP == 1) way_current = lfu_policy(index_current, 4);
                        else if (RP_MP == 2) way_current = random_policy(4);
                        else { cout << "Unknown RP" << endl; assert(0); }
                    }
                    irip_s4[index_current][way_current].vpn = previous_vpn & TAG_MASK;
                    irip_s4[index_current][way_current].successor_delta[0] = irip_s2[index][way].successor_delta[0];
                    irip_s4[index_current][way_current].successor_delta[1] = irip_s2[index][way].successor_delta[1];
                    irip_s4[index_current][way_current].successor_delta[2] = delta;
                    irip_s4[index_current][way_current].successor_delta[3] = 0;
                    irip_s4[index_current][way_current].confidence[0] = irip_s2[index][way].confidence[0];
                    irip_s4[index_current][way_current].confidence[1] = irip_s2[index][way].confidence[1];
                    irip_s4[index_current][way_current].confidence[2] = 0;
                    irip_s4[index_current][way_current].confidence[3] = 0;
                    irip_s4[index_current][way_current].freq = irip_s2[index][way].freq;
                    irip_s4[index_current][way_current].timestamp = irip_s2[index][way].timestamp;
                    irip_s2[index][way].timestamp = 0; irip_s2[index][way].freq = 0;
                    irip_s2[index][way].vpn = 0;
                    irip_s2[index][way].successor_delta[0] = 0; irip_s2[index][way].successor_delta[1] = 0;
                    irip_s2[index][way].confidence[0] = 0;      irip_s2[index][way].confidence[1] = 0;
                }
            }
        } else if (hit_level == 4) {
            for (int j = 0; j < 4; ++j) if (irip_s4[index][way].successor_delta[j] == delta) { bypass = 1; break; }
            if (!bypass) {
                for (int j = 0; j < 4; ++j)
                    if (irip_s4[index][way].successor_delta[j] == 0) {
                        irip_s4[index][way].successor_delta[j] = delta;
                        irip_s4[index][way].confidence[j] = 0;
                        placed = 1; break;
                    }
                if (!placed) {
                    bits_current  = (int)ceil(log2(PT_S8_SETS / PT_S8_ASSOC));
                    index_current = hash32((previous_vpn & TAG_MASK)) & ((1 << bits_current) - 1);
                    way_current   = search_markov_table(index_current, (previous_vpn & TAG_MASK), 8);
                    if (way_current == -1) {
                        if      (RP_MP == 0) way_current = lru_policy(index_current, 8);
                        else if (RP_MP == 1) way_current = lfu_policy(index_current, 8);
                        else if (RP_MP == 2) way_current = random_policy(8);
                        else { cout << "Unknown RP" << endl; assert(0); }
                    }
                    irip_s8[index_current][way_current].vpn = previous_vpn & TAG_MASK;
                    for (int k = 0; k < 4; k++) {
                        irip_s8[index_current][way_current].successor_delta[k] = irip_s4[index][way].successor_delta[k];
                        irip_s8[index_current][way_current].confidence[k]      = irip_s4[index][way].confidence[k];
                    }
                    irip_s8[index_current][way_current].successor_delta[4] = delta;
                    for (int k = 5; k < 8; k++) { irip_s8[index_current][way_current].successor_delta[k] = 0; irip_s8[index_current][way_current].confidence[k] = 0; }
                    irip_s8[index_current][way_current].confidence[4] = 0;
                    irip_s8[index_current][way_current].freq      = irip_s4[index][way].freq;
                    irip_s8[index_current][way_current].timestamp = irip_s4[index][way].timestamp;
                    irip_s4[index][way].timestamp = 0; irip_s4[index][way].freq = 0; irip_s4[index][way].vpn = 0;
                    for (int k = 0; k < 4; k++) { irip_s4[index][way].successor_delta[k] = 0; irip_s4[index][way].confidence[k] = 0; }
                }
            }
        } else if (hit_level == 8) {
            for (int j = 0; j < 8; ++j) if (irip_s8[index][way].successor_delta[j] == delta) { bypass = 1; break; }
            if (!bypass) {
                for (int j = 0; j < 8; ++j)
                    if (irip_s8[index][way].successor_delta[j] == 0) {
                        irip_s8[index][way].successor_delta[j] = delta;
                        irip_s8[index][way].confidence[j] = 0;
                        placed = 1; break;
                    }
                if (!placed) {
                    if (RP_MP == 1) {
                        int min_conf = irip_s8[index][way].confidence[0], min_pos = 0;
                        for (int l = 1; l < 8; ++l)
                            if (irip_s8[index][way].confidence[l] < min_conf) { min_conf = irip_s8[index][way].confidence[l]; min_pos = l; }
                        irip_s8[index][way].successor_delta[min_pos] = delta;
                        irip_s8[index][way].confidence[min_pos] = 0;
                    } else {
                        irip_s8[index][way].successor_delta[irip_s8[index][way].flag] = delta;
                        irip_s8[index][way].flag = (irip_s8[index][way].flag + 1) % 8;
                    }
                }
            }
        }

        // ── Update T-IRIP tables with this edge ───────────────────────────────
        tage_update(previous_vpn & TAG_MASK, delta);
    }

    // ── Update miss history register (push current to front) ─────────────────
    for (int i = TAGE_MAX_HIST - 1; i > 0; i--)
        history_reg[i] = history_reg[i-1];
    history_reg[0] = partial_vpn;
    if (history_len < TAGE_MAX_HIST) history_len++;

    // ── RLFU periodic frequency reset ────────────────────────────────────────
    if (decay_timer > RESET_FREQ) {
        reset_frequency();
        decay_timer = 0;
    }

    previous_vpn = current_vpn;
}

void CACHE::stlb_prefetcher_cache_fill(uint64_t addr, uint32_t set, uint32_t way,
                                        uint8_t prefetch, uint64_t evicted_addr) {}
